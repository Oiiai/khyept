#include <iostream>
#include <string>
using namespace std;

struct student {
    int id;
    int age;
    string name;
} s;

int main() {

    string ln;
    getline(cin, ln); s.id = stoi(ln);
    getline(cin, ln); s.age = stoi(ln);
    getline(cin, s.name);
    cout << s.id   << endl;
    cout << s.age  << endl;
    cout << s.name << endl;

    return 0;
}
